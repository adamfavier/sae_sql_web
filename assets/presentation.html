Requête A
SELECT Clientele.Societe, Clientele.Contact, Clientele.Tel, Clientele.MailCli
FROM Clientele
WHERE Clientele.Ville="Paris"
ORDER BY Clientele.Societe;

Requête B
SELECT Produit.NomProd, Produit.CoutAchat, Produit.CodeCateg, Produit.NoFour
FROM Pays INNER JOIN (Fournisseur INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour) ON Pays.NoPays = Fournisseur.NoPays
WHERE (((Produit.CoutAchat)>50) AND ((Produit.CodeCateg) Like "*8*" OR (Produit.CodeCateg) Like "*6*"))
ORDER BY Produit.NomProd;

Requête C
SELECT F.Societe, Pa.PaysFR, P.NomProd
FROM (((PAYS as Pa INNER JOIN FOURNISSEUR as F ON Pa.NoPays = F.NoPays)
INNER JOIN PRODUIT AS P ON F.NoFour = P.NoFour)
INNER JOIN CATEGORIE AS C ON P.CodeCateg = C.CodeCateg)
WHERE C.NomCateg LIKE "*Viandes*" OR C.NomCateg LIKE "*Poissons*" AND P.CoutAchat > 50
ORDER BY F.Societe, Pa.PaysFR, P.NomProd ASC;

Requête D
SELECT DISTINCT F.Societe, Ca.NomCateg
FROM ((CATEGORIE AS Ca INNER JOIN PRODUIT AS P ON Ca.CodeCateg = P.CodeCateg)
INNER JOIN FOURNISSEUR AS F ON P.NoFour = F.NoFour)
WHERE Ca.NomCateg LIKE "*Viande*" OR Ca.NomCateg LIKE "*Poissons*"
ORDER BY Ca.NomCateg;

Requête E
SELECT F.Societe
FROM ((CATEGORIE AS Ca INNER JOIN PRODUIT AS P ON Ca.CodeCateg = P.CodeCateg)
INNER JOIN FOURNISSEUR AS F ON P.NoFour = F.NoFour)
WHERE Ca.NomCateg LIKE "*Viandes*" AND Ca.NomCateg LIKE "*Poissons*"
ORDER BY F.Societe;

Requête F
SELECT P.NomProd
FROM Produit AS P
WHERE P.UnitStock IN ("carton", "bouteilles")
ORDER BY P.NomProd;

Requête G
SELECT TOP 3 Cl.Contact, Produit.NomProd
FROM Produit INNER JOIN ((CLIENTELE AS Cl INNER JOIN Commande ON Cl.CodeCli = Commande.CodeCli)
INNER JOIN DetailCommande ON Commande.[N° commande] = DetailCommande.[N° commande])
ON Produit.RefProd = DetailCommande.[Réf produit]
WHERE (((Cl.Contact)="Max Deschamps"));

Requête H
SELECT DISTINCT F.Societe, P.NomProd
FROM ((FOURNISSEUR AS F INNER JOIN PRODUIT AS P ON F.NoFour = P.NoFour)
INNER JOIN CATEGORIE AS C ON P.CodeCateg = C.CodeCateg)
WHERE C.NomCateg = "Desserts";

Requête I
SELECT Fournisseur.NoFour, Fournisseur.Societe, Produit.NomProd, Pays.PaysFR
FROM Pays INNER JOIN (Fournisseur INNER JOIN Produit ON Fournisseur.NoFour = Produit.NoFour) ON Pays.NoPays = Fournisseur.NoPays
WHERE (((Pays.[PaysFR]) Not In ('France')) AND ((Produit.NoFour) Is Null));

Requête J
SELECT Clientele.CodeCli
FROM Clientele INNER JOIN Clientele AS Clientele_1 ON Clientele.CodeCli = Clientele_1.Ville
WHERE ((Clientele_1.Ville)="Paris") AND (Clientele.CodeCli = Clientele_1.CodeCliPart)
ORDER BY Clientele.CodeCli;

Requête 2
SELECT SUM(P.CoutAchat * DC.Quantité * (1 - DC.[Remise (%)])) AS chiffre_affaires
FROM (((PAYS as Pa INNER JOIN FOURNISSEUR AS F ON Pa.NoPays = F.NoPays)
INNER JOIN PRODUIT AS P ON F.NoFour = P.NoFour)
INNER JOIN DETAILCOMMANDE AS DC ON P.RefProd = DC.[Réf produit])
WHERE UE = "Membre";

Requête 3
SELECT SUM(P.CoutAchat * DC.Quantité * (1 - DC.[Remise (%)])) AS chiffre_affaires
FROM (((Commande as C INNER JOIN DetailCommande AS DC ON C.[N° commande] = DC.[N° commande])
INNER JOIN Produit as P ON DC.[Réf produit] = P.RefProd)
INNER JOIN Categorie AS Ca ON P.CodeCateg = Ca.CodeCateg)
WHERE Ca.NomCateg LIKE "*Boissons*" OR Ca.NomCateg LIKE "*Desserts*" AND C.DateCde BETWEEN #31/12/2021# AND #01/01/2023#;

Requête 4
SELECT SUM(P.UnitStock) AS nombre_de_produits
FROM Categorie as C INNER JOIN Produit as P ON C.CodeCateg = P.CodeCateg
WHERE C.NomCateg LIKE "*Viandes*" AND P.UnitStock > 0;
